import { SourceReport, Study } from "./types";

/**
 * For a real dataset study, `study.sourceReport` is the only thing this
 * panel is allowed to show â€” when it's undefined, no report was matched and
 * the panel must say so, never synthesize one.
 *
 * For a mock/uploaded study (no `datasetSource`), there is no separate
 * "source" report concept â€” the working report IS the only report that has
 * ever existed for it. Mirroring its already-visible content here (read
 * only) is not fabrication, just a second read-only view of the same
 * existing data, so the panel stays useful in mock mode instead of always
 * saying "not available".
 */
export function resolveReportPanelContent(study: Study): SourceReport | undefined {
  if (study.sourceReport) return study.sourceReport;
  if (study.datasetSource) return undefined; // real dataset, genuinely no report matched

  const s = study.report.sections;
  if (!s.clinicalInformation && !s.technique && !s.findings && !s.impression) return undefined;
  return {
    source: "Meridian",
    clinicalInformation: s.clinicalInformation || undefined,
    technique: s.technique || undefined,
    findings: s.findings || undefined,
    impression: s.impression || undefined,
    text: [s.clinicalInformation, s.technique, s.findings, s.impression].filter(Boolean).join("\n\n"),
  };
}

import { DatasetStudy, DATASET_SOURCE_LABEL } from "./types";
import { Report, SeriesInfo, SourceReport, Study } from "@/lib/types";

/** Builds the URL the browser will actually load pixels from. Resolved server-side against the manifest â€” never a raw filesystem path. */
export function imageUrlFor(studyId: string, seriesId: string, index: number): string {
  return `/api/studies/${encodeURIComponent(studyId)}/images/${encodeURIComponent(seriesId)}/${index}`;
}

function numericSeed(id: string): number {
  return parseInt(id.slice(0, 8), 16) || 0;
}

function toSourceReport(ds: DatasetStudy): SourceReport | undefined {
  if (!ds.report) return undefined;
  return {
    source: ds.report.source,
    text: ds.report.text,
    clinicalInformation: ds.report.clinicalInformation,
    indication: ds.report.indication,
    technique: ds.report.technique,
    findings: ds.report.findings,
    impression: ds.report.impression,
    comparison: ds.report.comparison,
  };
}

/**
 * Seeds the editable "working report" from the source report's structured
 * sections where available, falling back to the full original text in
 * `findings` when the source report is unstructured free text. This is the
 * radiologist's starting point for editing â€” the immutable original always
 * stays available separately via `study.sourceReport`.
 */
function toWorkingReport(ds: DatasetStudy): Report {
  const src = ds.report;
  return {
    status: src ? "in-review" : "ai-draft",
    sections: {
      clinicalInformation: src?.clinicalInformation ?? src?.indication ?? "",
      technique: src?.technique ?? "",
      findings: src?.findings ?? (src && !src.clinicalInformation && !src.findings && !src.impression ? src.text : "") ?? "",
      impression: src?.impression ?? "",
    },
    aiAssisted: false,
    version: 1,
    reviewer: null,
    approvedAt: null,
  };
}

export function toAppStudy(ds: DatasetStudy): Study {
  const series: SeriesInfo[] = ds.series.map((s) => ({
    id: s.id,
    label: s.label,
    sliceCount: s.sliceCount,
    orientation: s.orientation,
    images: s.images.map((img) => ({
      index: img.index,
      imageUrl: imageUrlFor(ds.id, s.id, img.index),
      width: img.width,
      height: img.height,
    })),
  }));

  return {
    id: ds.id,
    studyId: ds.studyCode,
    patientAlias: ds.patientAlias,
    age: ds.age,
    sex: ds.sex,
    modality: ds.modality,
    bodyRegion: ds.bodyRegion,
    date: ds.date,
    priority: "routine", // never inferred/fabricated from clinical content
    status: ds.report ? "in-review" : "queued",
    aiConfidence: null,
    assignedRadiologist: null,
    series,
    findings: (ds.aiFindings ?? []).map((f, i) => ({
      id: `${ds.id}-ai-${i}`,
      label: f.label,
      confidence: f.confidence,
      band: f.band,
      region: f.region,
      sliceIndex: f.sliceIndex,
      seriesId: f.seriesId,
      description: f.description,
      status: "pending" as const,
    })),
    thumbnailSeed: numericSeed(ds.id),
    report: toWorkingReport(ds),
    datasetSource: DATASET_SOURCE_LABEL[ds.dataset],
    sourceReport: toSourceReport(ds),
    aiStatus: ds.aiFindings && ds.aiFindings.length > 0 ? "complete" : "not-run",
  };
}

/**
 * ============================================================================
 * MRI DATASET ADAPTER
 * ============================================================================
 *
 * Supported:
 *
 *   MR-RATE:
 *     - ZIP -> NIfTI volumes
 *     - .nii
 *     - .nii.gz
 *
 *   Generic MRI:
 *     - DICOM
 *     - NIfTI
 *     - PNG/JPG/WEBP
 *
 * Important:
 *
 * Do NOT rely only on path.extname() for NIfTI.
 *
 * path.extname("volume.nii.gz") === ".gz"
 *
 * Therefore we explicitly check:
 *
 *   .nii
 *   .nii.gz
 *
 * using endsWith().
 * ============================================================================
 */

const MRI_IMAGE_EXTENSIONS = [
  ...DICOM_EXTS,
  ...RASTER_EXTS,
  ".nii",
  ".nii.gz",
];

const IGNORED_DIR_NAMES = new Set([
  "_inspection",
  "_scripts",
  "_cache",
  "inspection",
  "metadata",
  "reports",
  "splits",
  "seg",
  "masks",
  "mask",
]);

const IGNORED_PATH_PARTS = [
  `${path.sep}seg${path.sep}`,
  `${path.sep}masks${path.sep}`,
  `${path.sep}mask${path.sep}`,
  `${path.sep}defacing${path.sep}`,
];

/**
 * True when a file is a supported MRI display file.
 *
 * This deliberately uses endsWith() instead of path.extname()
 * so .nii.gz works correctly.
 */
function isSupportedMRIImage(file: string): boolean {
  const lower = file.toLowerCase();

  return MRI_IMAGE_EXTENSIONS.some((ext) =>
    lower.endsWith(ext.toLowerCase())
  );
}

/**
 * Returns all supported MRI image files recursively.
 *
 * We intentionally do not depend on walk(... extensions)
 * because .nii.gz can be incorrectly interpreted as .gz.
 */
function findMRIImageFiles(dir: string): string[] {
  const results: string[] = [];

  function visit(currentDir: string) {
    let entries: fs.Dirent[];

    try {
      entries = fs.readdirSync(currentDir, {
        withFileTypes: true,
      });
    } catch {
      return;
    }

    for (const entry of entries) {
      const fullPath = path.join(currentDir, entry.name);

      if (entry.isDirectory()) {
        const lowerName = entry.name.toLowerCase();

        if (
          lowerName === "seg" ||
          lowerName === "mask" ||
          lowerName === "masks" ||
          lowerName === "defacing"
        ) {
          continue;
        }

        visit(fullPath);
        continue;
      }

      if (entry.isFile() && isSupportedMRIImage(fullPath)) {
        results.push(fullPath);
      }
    }
  }

  visit(dir);

  return results.sort(naturalSort);
}

/**
 * Determine whether a directory should be treated as a loose MRI study.
 */
function isValidLooseStudyDir(dir: string): boolean {
  const name = path.basename(dir).toLowerCase();

  if (IGNORED_DIR_NAMES.has(name)) {
    return false;
  }

  if (name.startsWith(".")) {
    return false;
  }

  return true;
}

export const mriAdapter: DatasetAdapter = {
  id: "MRI",

  matchesDir: ["03_MRI"],

  async scan(rootDir): Promise<AdapterResult> {
    const issues: AdapterIssue[] = [];

    if (!safeExists(rootDir)) {
      return {
        studies: [],
        issues,
      };
    }

    // ------------------------------------------------------------------------
    // Load metadata and reports
    // ------------------------------------------------------------------------

    const csvFiles = [...walk(rootDir, { extensions: [".csv"] })];

    const { metaLookup, reportLookup } = loadTables(
      csvFiles,
      issues
    );

    // ------------------------------------------------------------------------
    // MR-RATE ZIP files
    // ------------------------------------------------------------------------

    const zipFiles = [...walk(rootDir, { extensions: [".zip"] })]
      .filter((file) => {
        const lower = file.toLowerCase();

        // Ignore ZIPs from unrelated cache/backup folders if any.
        return !lower.includes(`${path.sep}cache${path.sep}zips${path.sep}`);
      })
      .sort(naturalSort);

    const studies: DatasetStudy[] = [];

    for (const zip of zipFiles) {
      const stem = path.basename(
        zip,
        path.extname(zip)
      );

      let extractedDir: string;

      try {
        extractedDir = ensureExtracted(zip);
      } catch (err) {
        issues.push({
          dataset: "MRI",
          path: zip,
          reason:
            `Failed to extract ZIP: ${
              err instanceof Error
                ? err.message
                : String(err)
            }`,
        });

        continue;
      }

      // --------------------------------------------------------------
      // IMPORTANT:
      //
      // Search manually for .nii.gz.
      //
      // This fixes the primary MRI problem.
      // --------------------------------------------------------------

      const imageFiles = findMRIImageFiles(
        extractedDir
      );

      if (imageFiles.length === 0) {
        issues.push({
          dataset: "MRI",
          path: extractedDir,
          reason:
            "ZIP extracted successfully but no supported MRI image files were found. Checked .nii, .nii.gz, .dcm, .png, .jpg, .jpeg and .webp.",
        });

        continue;
      }

      const study = buildStudyFromDir(
        extractedDir,
        stem,
        metaLookup,
        reportLookup,
        rootDir,
        issues
      );

      if (study) {
        studies.push(study);
      }
    }

    // ------------------------------------------------------------------------
    // Loose MRI study directories
    // ------------------------------------------------------------------------

    const topLevelDirs = safeListStudyDirs(rootDir);

    for (const dir of topLevelDirs) {
      if (!isValidLooseStudyDir(dir)) {
        continue;
      }

      const imageFiles = findMRIImageFiles(dir);

      if (imageFiles.length === 0) {
        continue;
      }

      const stem = path.basename(dir);

      const study = buildStudyFromDir(
        dir,
        stem,
        metaLookup,
        reportLookup,
        rootDir,
        issues
      );

      if (study) {
        studies.push(study);
      }
    }

    // ------------------------------------------------------------------------
    // Deduplicate studies
    // ------------------------------------------------------------------------

    const uniqueStudies = new Map<string, DatasetStudy>();

    for (const study of studies) {
      if (!uniqueStudies.has(study.id)) {
        uniqueStudies.set(study.id, study);
      }
    }

    return {
      studies: [...uniqueStudies.values()],
      issues,
    };
  },
};

/**
 * ============================================================================
 * TOP-LEVEL DIRECTORY DISCOVERY
 * ============================================================================
 */

function safeListStudyDirs(
  rootDir: string
): string[] {
  try {
    return fs
      .readdirSync(rootDir, {
        withFileTypes: true,
      })
      .filter((entry) => entry.isDirectory())
      .map((entry) =>
        path.join(rootDir, entry.name)
      );
  } catch {
    return [];
  }
}

/**
 * ============================================================================
 * BUILD STUDY
 * ============================================================================
 */

function buildStudyFromDir(
  dir: string,
  idHint: string,
  metaLookup: Map<string, MetaRow>,
  reportLookup: Map<string, ReportRow>,
  rootDir: string,
  issues: AdapterIssue[]
): DatasetStudy | undefined {
  // --------------------------------------------------------------------------
  // Find image files first.
  //
  // This is the critical change.
  // --------------------------------------------------------------------------

  const allImageFiles = findMRIImageFiles(dir);

  if (allImageFiles.length === 0) {
    issues.push({
      dataset: "MRI",
      path: dir,
      reason:
        "No supported MRI image files found. Expected .nii/.nii.gz, .dcm, .png, .jpg, .jpeg or .webp.",
    });

    return undefined;
  }

  // --------------------------------------------------------------------------
  // Identify actual series directories.
  //
  // MR-RATE commonly has:
  //
  //   img/
  //   seg/
  //
  // We use img as primary imaging and ignore seg.
  // --------------------------------------------------------------------------

  const candidateSeriesDirs = findSeriesDirectories(
    dir,
    allImageFiles
  );

  const series: DatasetSeries[] = [];

  for (
    let i = 0;
    i < candidateSeriesDirs.length;
    i++
  ) {
    const seriesDir = candidateSeriesDirs[i];

    const files = findMRIImageFiles(
      seriesDir
    );

    if (files.length === 0) {
      continue;
    }

    const lowerSeriesPath =
      seriesDir.toLowerCase();

    // --------------------------------------------------------------
    // Never expose segmentation/mask data as primary MRI series.
    // --------------------------------------------------------------

    if (
      lowerSeriesPath.includes(
        `${path.sep}seg${path.sep}`
      ) ||
      lowerSeriesPath.endsWith(
        `${path.sep}seg`
      ) ||
      lowerSeriesPath.includes(
        `${path.sep}mask${path.sep}`
      ) ||
      lowerSeriesPath.includes(
        `${path.sep}masks${path.sep}`
      )
    ) {
      continue;
    }

    // --------------------------------------------------------------
    // Detect orientation.
    // --------------------------------------------------------------

    let orientation:
      DatasetSeries["orientation"] = "Axial";

    // DICOM orientation
    for (const file of files) {
      if (
        file.toLowerCase().endsWith(".dcm")
      ) {
        try {
          const summary =
            readDicomSummary(file);

          if (summary?.orientation) {
            orientation =
              summary.orientation as DatasetSeries["orientation"];
          }
        } catch {
          // Continue with filename inference.
        }

        break;
      }
    }

    // NIfTI filename inference
    for (const file of files) {
      const lowerName =
        path.basename(file).toLowerCase();

      if (
        lowerName.includes("-axi") ||
        lowerName.includes("_axi") ||
        lowerName.includes("axial")
      ) {
        orientation = "Axial";
        break;
      }

      if (
        lowerName.includes("-cor") ||
        lowerName.includes("_cor") ||
        lowerName.includes("coronal")
      ) {
        orientation = "Coronal";
        break;
      }

      if (
        lowerName.includes("-sag") ||
        lowerName.includes("_sag") ||
        lowerName.includes("sagittal")
      ) {
        orientation = "Sagittal";
        break;
      }
    }

    // --------------------------------------------------------------
    // Stable series ID
    // --------------------------------------------------------------

    const seriesName =
      path.basename(seriesDir);

    series.push(
      buildSeriesFromFiles({
        seriesId: stableId(
          "MRI",
          idHint,
          seriesName,
          String(i)
        ),

        label:
          seriesName ===
          path.basename(dir)
            ? `Series ${i + 1}`
            : seriesName,

        modality: "MRI",

        orientation,

        files,
      })
    );
  }

  // --------------------------------------------------------------------------
  // Safety fallback:
  //
  // If the directory itself contains NIfTI files and no child series was
  // detected, create one series from the entire directory.
  // --------------------------------------------------------------------------

  if (series.length === 0) {
    const directFiles =
      allImageFiles.filter(
        (file) =>
          path.dirname(file) === dir
      );

    if (directFiles.length > 0) {
      series.push(
        buildSeriesFromFiles({
          seriesId: stableId(
            "MRI",
            idHint,
            "main"
          ),

          label: "MRI",

          modality: "MRI",

          orientation: inferOrientation(
            directFiles
          ),

          files: directFiles,
        })
      );
    }
  }

  // --------------------------------------------------------------------------
  // No usable series
  // --------------------------------------------------------------------------

  if (series.length === 0) {
    issues.push({
      dataset: "MRI",
      path: dir,
      reason:
        "MRI image files were found but no primary displayable series could be constructed.",
    });

    return undefined;
  }

  // --------------------------------------------------------------------------
  // Metadata
  // --------------------------------------------------------------------------

  const meta = matchByIdentifiers(
    metaLookup,
    buildIdentifierCandidates(
      idHint,
      dir
    )
  );

  // --------------------------------------------------------------------------
  // Report
  // --------------------------------------------------------------------------

  const reportRow =
    matchByIdentifiers(
      reportLookup,
      buildIdentifierCandidates(
        idHint,
        dir
      )
    );

  const report: StudyReport | undefined =
    reportRow
      ? {
          source: "MR-RATE",
          text: reportRow.text,
        }
      : undefined;

  // --------------------------------------------------------------------------
  // Stable study ID
  // --------------------------------------------------------------------------

  const id = stableId(
    "MRI",
    idHint
  );

  return {
    id,

    dataset: "MRI",

    studyCode: studyCode(
      "MRI",
      idHint
    ),

    sourceStudyId: idHint,

    patientAlias:
      `Patient ${id
        .slice(0, 6)
        .toUpperCase()}`,

    age: meta?.age,

    sex: meta?.sex,

    modality: "MRI",

    bodyRegion: meta?.bodyRegion,

    date: meta?.date,

    series,

    report,

    sourceDir: rootDir,
  };
}

/**
 * ============================================================================
 * FIND SERIES DIRECTORIES
 * ============================================================================
 *
 * If the extracted ZIP contains:
 *
 *   img/
 *   seg/
 *
 * return img.
 *
 * If it contains:
 *
 *   t1/
 *   t2/
 *   flair/
 *
 * return those directories.
 *
 * If images are directly inside the root, return root.
 * ============================================================================
 */

function findSeriesDirectories(
  studyDir: string,
  imageFiles: string[]
): string[] {
  const directChildDirs: string[] = [];

  try {
    const entries =
      fs.readdirSync(studyDir, {
        withFileTypes: true,
      });

    for (const entry of entries) {
      if (!entry.isDirectory()) {
        continue;
      }

      const name =
        entry.name.toLowerCase();

      if (
        name === "seg" ||
        name === "mask" ||
        name === "masks" ||
        name === "defacing"
      ) {
        continue;
      }

      directChildDirs.push(
        path.join(
          studyDir,
          entry.name
        )
      );
    }
  } catch {
    return [studyDir];
  }

  // Direct image files -> study root is itself a series.
  const directImages =
    imageFiles.filter(
      (file) =>
        path.dirname(file) === studyDir
    );

  if (
    directImages.length > 0 &&
    directChildDirs.length === 0
  ) {
    return [studyDir];
  }

  // If child directories exist, use them.
  if (directChildDirs.length > 0) {
    return directChildDirs;
  }

  return [studyDir];
}

/**
 * ============================================================================
 * ORIENTATION INFERENCE
 * ============================================================================
 */

function inferOrientation(
  files: string[]
): DatasetSeries["orientation"] {
  for (const file of files) {
    const name =
      path.basename(file).toLowerCase();

    if (
      name.includes("-axi") ||
      name.includes("_axi") ||
      name.includes("axial")
    ) {
      return "Axial";
    }

    if (
      name.includes("-cor") ||
      name.includes("_cor") ||
      name.includes("coronal")
    ) {
      return "Coronal";
    }

    if (
      name.includes("-sag") ||
      name.includes("_sag") ||
      name.includes("sagittal")
    ) {
      return "Sagittal";
    }
  }

  return "Axial";
}

/**
 * ============================================================================
 * IDENTIFIER CANDIDATES
 * ============================================================================
 */

function buildIdentifierCandidates(
  idHint: string,
  dir: string
): string[] {
  const candidates = new Set<string>();

  const add = (value?: string) => {
    if (!value) return;

    const clean =
      value.trim();

    if (!clean) return;

    candidates.add(clean);
    candidates.add(clean.toLowerCase());
    candidates.add(clean.toUpperCase());
  };

  add(idHint);
  add(path.basename(dir));

  // Remove common filename suffixes.
  const withoutZip =
    idHint.replace(
      /\.zip$/i,
      ""
    );

  add(withoutZip);

  return [...candidates];
}

/**
 * ============================================================================
 * CSV LOADING
 * ============================================================================
 */

function loadTables(
  csvFiles: string[],
  issues: AdapterIssue[]
) {
  const metaRows: MetaRow[] = [];

  const reportRows: ReportRow[] = [];

  for (const csv of csvFiles) {
    const content =
      readTextFileSafe(csv);

    if (!content) {
      issues.push({
        dataset: "MRI",
        path: csv,
        reason: "Unreadable CSV file",
      });

      continue;
    }

    let parsed:
      Record<string, string>[];

    try {
      parsed = parseCsv(content);
    } catch {
      issues.push({
        dataset: "MRI",
        path: csv,
        reason: "Malformed CSV",
      });

      continue;
    }

    if (parsed.length === 0) {
      continue;
    }

    const looksLikeReports =
      !!pickColumn(
        parsed[0],
        [
          "report",
          "findings",
          "impression",
          "text",
        ]
      );

    for (const row of parsed) {
      // --------------------------------------------------------------
      // MR-RATE identifiers
      // --------------------------------------------------------------

      const key =
        pickColumn(
          row,
          [
            "study_uid",
            "study_id",
            "studyid",
            "case_id",
            "caseid",
            "batch_id",
            "id",
            "filename",
            "file_name",
            "zip",
            "zip_name",
            "accession",
            "accession_number",
          ]
        );

      if (looksLikeReports) {
        const text =
          pickColumn(
            row,
            [
              "report",
              "findings",
              "impression",
              "text",
              "caption",
              "description",
            ]
          );

        if (
          key &&
          text
        ) {
          reportRows.push({
            key: key.trim(),
            text: text.trim(),
          });
        }

        continue;
      }

      // --------------------------------------------------------------
      // Metadata
      // --------------------------------------------------------------

      const ageStr =
        pickColumn(
          row,
          [
            "age",
            "patient_age",
          ]
        );

      const sexStr =
        pickColumn(
          row,
          [
            "sex",
            "gender",
          ]
        );

      const normalizedSex =
        sexStr?.trim().toUpperCase();

      let sex:
        | "M"
        | "F"
        | "Other"
        | undefined;

      if (
        normalizedSex?.startsWith("M")
      ) {
        sex = "M";
      } else if (
        normalizedSex?.startsWith("F")
      ) {
        sex = "F";
      } else if (
        normalizedSex
      ) {
        sex = "Other";
      }

      let age:
        | number
        | undefined;

      if (ageStr) {
        const parsedAge =
          Number.parseInt(
            ageStr,
            10
          );

        if (
          Number.isFinite(
            parsedAge
          )
        ) {
          age = parsedAge;
        }
      }

      metaRows.push({
        key: key?.trim(),

        age,

        sex,

        bodyRegion:
          pickColumn(
            row,
            [
              "body_region",
              "region",
              "anatomy",
              "bodypart",
              "body_part",
            ]
          )?.trim(),

        date:
          pickColumn(
            row,
            [
              "date",
              "study_date",
              "exam_date",
              "anon_study_date",
            ]
          )?.trim(),

        raw: row,
      });
    }
  }

  return {
    metaLookup:
      buildLookup(
        metaRows,
        (row) => row.key
      ),

    reportLookup:
      buildLookup(
        reportRows,
        (row) => row.key
      ),
  };
}

  DatasetAdapter,

  key?: string;
  text: string;

export const ctAdapter: DatasetAdapter = {
  id: "CT",
  matchesDir: ["02_CT"],

  async scan(rootDir): Promise<AdapterResult> {
    const issues: AdapterIssue[] = [];

    if (!safeExists(rootDir)) {
      return { studies: [], issues };
    }

    /*
     * Perle CT contains DICOM files with NO .dcm extension.
     */
    const candidateFiles = [...walk(rootDir)];

    /*
     * Load CSV/JSON reports plus Perle TXT reports.
     */
    const csvFiles = [
      ...walk(rootDir, {
        extensions: [".csv"],
      }),
    ];

    const jsonFiles = [
      ...walk(rootDir, {
        extensions: [".json"],
      }),
    ];

    const txtFiles = [
      ...walk(rootDir, {
        extensions: [".txt"],
      }),
    ];

    const reportLookup = loadReports(
      csvFiles,
      jsonFiles,
      txtFiles,
      issues
    );

    /*
     * studyKey -> seriesKey -> DICOM instances
     */
    type Instance = {
      absPath: string;
      summary: ReturnType<typeof readDicomSummary>;
    };

    const studies = new Map<
      string,
      Map<string, Instance[]>
    >();

    /*
     * Study-level metadata.
     */
    const studyMeta = new Map<
      string,
      {
        accession?: string;
        bodyPart?: string;
        date?: string;
      }
    >();

    /*
     * Read every candidate file.
     */
    for (const file of candidateFiles) {
      const summary = readDicomSummary(file);

      if (!summary) {
        continue;
      }

      /*
       * Prefer DICOM StudyInstanceUID.
       *
       * If unavailable, fall back to the dataset
       * folder structure:
       *
       * PA0_ct_chest/ST0/SE2/IM0
       */
      const studyKey =
        summary.studyInstanceUID ||
        path.basename(
          path.dirname(
            path.dirname(file)
          )
        );

      /*
       * Prefer DICOM SeriesInstanceUID.
       */
      const seriesKey =
        summary.seriesInstanceUID ||
        path.basename(
          path.dirname(file)
        );

      if (!studies.has(studyKey)) {
        studies.set(
          studyKey,
          new Map<string, Instance[]>()
        );
      }

      const seriesMap =
        studies.get(studyKey)!;

      if (!seriesMap.has(seriesKey)) {
        seriesMap.set(
          seriesKey,
          []
        );
      }

      seriesMap
        .get(seriesKey)!
        .push({
          absPath: file,
          summary,
        });

      /*
       * Save study metadata once.
       */
      if (!studyMeta.has(studyKey)) {
        studyMeta.set(
          studyKey,
          {
            accession:
              summary.accessionNumber,

            bodyPart:
              summary.bodyPart,

            date:
              normalizeDicomDate(
                summary.studyDate
              ),
          }
        );
      }
    }

    /*
     * Convert grouped DICOM files into DatasetStudy objects.
     */
    const result: DatasetStudy[] = [];

    for (const [
      studyKey,
      seriesMap,
    ] of studies) {
      const series: DatasetSeries[] = [];

      let seriesIdx = 0;

      for (const [
        seriesKey,
        instances,
      ] of seriesMap) {
        /*
         * Sort CT slices.
         */
        instances.sort(
          (a, b) => {
            const an =
              a.summary?.instanceNumber;

            const bn =
              b.summary?.instanceNumber;

            if (
              an !== undefined &&
              bn !== undefined &&
              an !== bn
            ) {
              return an - bn;
            }

            const asl =
              a.summary?.sliceLocation;

            const bsl =
              b.summary?.sliceLocation;

            if (
              asl !== undefined &&
              bsl !== undefined &&
              asl !== bsl
            ) {
              return asl - bsl;
            }

            return naturalSort(
              a.absPath,
              b.absPath
            );
          }
        );

        /*
         * Convert DICOM instances to viewer images.
         */
        const images: DatasetImageRef[] =
          instances.map(
            (inst, index) => ({
              id: stableId(
                seriesKey,
                String(index)
              ),

              index,

              kind: "dicom",

              absPath:
                inst.absPath,

              width:
                inst.summary?.columns,

              height:
                inst.summary?.rows,
            })
          );

        seriesIdx++;

        series.push({
          id: stableId(
            studyKey,
            seriesKey
          ),

          label:
            instances[0]?.summary
              ?.seriesDescription ||
            `Series ${seriesIdx}`,

          modality: "CT",

          orientation: "Axial",

          sliceCount:
            images.length,

          images,
        });
      }

      if (series.length === 0) {
        continue;
      }

      const meta =
        studyMeta.get(studyKey);

      /*
       * Match report using:
       *
       * 1. StudyInstanceUID
       * 2. Accession
       * 3. Dataset folder/study ID
       *
       * The third identifier is important for
       * Perle CT because the report is associated
       * with PA0_ct_chest, PA1_ct_brain, etc.
       */
      const folderStudyKey =
        getDatasetStudyKey(
          studyKey,
          seriesMap
        );

      const reportRow =
        matchByIdentifiers(
          reportLookup,
          [
            studyKey,
            meta?.accession,
            folderStudyKey,
          ]
        );

      const report:
        | StudyReport
        | undefined =
        reportRow
          ? {
              source: "Perle CT",
              text: reportRow.text,
            }
          : undefined;

      const id =
        stableId(
          "CT",
          studyKey
        );

      result.push({
        id,

        dataset: "CT",

        studyCode:
          studyCode(
            "CT",
            studyKey
          ),

        sourceStudyId:
          studyKey,

        patientAlias:
          `Patient ${id
            .slice(0, 6)
            .toUpperCase()}`,

        modality: "CT",

        bodyRegion:
          meta?.bodyPart,

        date:
          meta?.date,

        series,

        report,

        sourceDir:
          rootDir,
      });
    }

    return {
      studies: result,
      issues,
    };
  },
};

/**
 * Convert DICOM YYYYMMDD into YYYY-MM-DD.
 */
function normalizeDicomDate(
  value?: string
): string | undefined {
  if (
    !value ||
    value.length !== 8
  ) {
    return undefined;
  }

  return (
    `${value.slice(0, 4)}-` +
    `${value.slice(4, 6)}-` +
    `${value.slice(6, 8)}`
  );
}

/**
 * Extract the dataset study identifier from
 * a Perle CT path.
 *
 * Example:
 *
 * .../ct_general/PA0_ct_chest/ST0/SE2/IM0
 *
 * returns:
 *
 * PA0_ct_chest
 */
function getDatasetStudyKey(
  studyKey: string,
  seriesMap: Map<string, any>
): string | undefined {
  for (const instances of seriesMap.values()) {
    const first =
      instances?.[0];

    if (!first?.absPath) {
      continue;
    }

    const normalized =
      first.absPath.replace(
        /\\/g,
        "/"
      );

    const marker =
      "/ct_general/";

    const markerIndex =
      normalized.indexOf(
        marker
      );

    if (markerIndex >= 0) {
      const remainder =
        normalized.slice(
          markerIndex +
            marker.length
        );

      const parts =
        remainder.split("/");

      if (parts.length > 0) {
        return parts[0];
      }
    }
  }

  /*
   * If the study key itself is already
   * a dataset identifier, use it.
   */
  if (
    /^PA\d+_ct_/i.test(
      studyKey
    )
  ) {
    return studyKey;
  }

  return undefined;
}

/**
 * Load CT reports from:
 *
 * CSV
 * JSON
 * TXT
 */
function loadReports(
  csvFiles: string[],
  jsonFiles: string[],
  txtFiles: string[],
  issues: AdapterIssue[]
): Map<string, ReportRow> {
  const rows: ReportRow[] = [];

  /*
   * CSV reports.
   */
  for (const csv of csvFiles) {
    const content =
      readTextFileSafe(csv);

    if (!content) {
      issues.push({
        dataset: "CT",
        path: csv,
        reason:
          "Unreadable CSV file",
      });

      continue;
    }

    let parsed:
      Record<string, string>[];

    try {
      parsed =
        parseCsv(content);
    } catch {
      issues.push({
        dataset: "CT",
        path: csv,
        reason:
          "Malformed CSV",
      });

      continue;
    }

    for (const row of parsed) {
      const key =
        pickColumn(
          row,
          [
            "studyinstanceuid",
            "study_instance_uid",
            "accession",
            "accession_number",
            "study_id",
            "subject_id",
            "id",
          ]
        );

      const text =
        pickColumn(
          row,
          [
            "report",
            "findings",
            "impression",
            "text",
            "narrative",
          ]
        );

      if (key && text) {
        rows.push({
          key,
          text,
          raw: row,
        });
      }
    }
  }

  /*
   * JSON reports.
   */
  for (const jsonFile of jsonFiles) {
    const content =
      readTextFileSafe(
        jsonFile
      );

    if (!content) {
      continue;
    }

    try {
      const data =
        JSON.parse(content);

      const entries =
        Array.isArray(data)
          ? data
          : Object.values(data);

      for (const entry of entries as any[]) {
        const key =
          entry.StudyInstanceUID ??
          entry.study_instance_uid ??
          entry.accession_number ??
          entry.study_id ??
          entry.subject_id ??
          entry.id;

        const text =
          entry.report ??
          entry.findings ??
          entry.impression ??
          entry.text;

        if (key && text) {
          rows.push({
            key: String(key),
            text: String(text),
            raw: entry,
          });
        }
      }
    } catch {
      issues.push({
        dataset: "CT",
        path: jsonFile,
        reason:
          "Malformed JSON",
      });
    }
  }

  /*
   * Perle CT TXT reports.
   *
   * Example:
   *
   * PA0_ct_chest/
   *   ...report.txt
   *
   * We derive the subject/study ID from
   * the directory structure.
   */
  for (const txtFile of txtFiles) {
    const content =
      readTextFileSafe(
        txtFile
      );

    if (!content?.trim()) {
      continue;
    }

    const key =
      getTxtStudyKey(
        txtFile
      );

    if (!key) {
      continue;
    }

    rows.push({
      key,
      text: content.trim(),
      raw: {
        sourceFile:
          txtFile,
      },
    });
  }

  return buildLookup(
    rows,
    (row) => row.key
  );
}

/**
 * Extract PA0_ct_chest / PA1_ct_brain
 * from a Perle CT report path.
 */
function getTxtStudyKey(
  txtFile: string
): string | undefined {
  const normalized =
    txtFile.replace(
      /\\/g,
      "/"
    );

  /*
   * Preferred structure:
   *
   * .../ct_general/PA0_ct_chest/report.txt
   */
  const marker =
    "/ct_general/";

  const markerIndex =
    normalized.indexOf(
      marker
    );

  if (markerIndex >= 0) {
    const remainder =
      normalized.slice(
        markerIndex +
          marker.length
      );

    const parts =
      remainder.split("/");

    if (
      parts.length > 0 &&
      /^PA\d+_ct_/i.test(
        parts[0]
      )
    ) {
      return parts[0];
    }
  }

  /*
   * Inspection reports:
   *
   * .../inspection/PA0_ct_chest/report.txt
   */
  const inspectionMarker =
    "/inspection/";

  const inspectionIndex =
    normalized.indexOf(
      inspectionMarker
    );

  if (inspectionIndex >= 0) {
    const remainder =
      normalized.slice(
        inspectionIndex +
          inspectionMarker.length
      );

    const parts =
      remainder.split("/");

    if (
      parts.length > 0 &&
      /^PA\d+_ct_/i.test(
        parts[0]
      )
    ) {
      return parts[0];
    }
  }

  return undefined;
}
